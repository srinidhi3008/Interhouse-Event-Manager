create database school;
use school;

create table info
(Admission_no integer Primary key,
 First_Name varchar(20),
 Last_Name varchar(20),
 Grade integer,
 Section char(1),
 Captain_ViceCaptain varchar(20) default "No",
 House varchar(20));

create table cred
(Username varchar(6) Primary key,
 Password varchar(20) NOT NULL);

create table events
(Event_Name varchar(25),
 Grades varchar(5),
 Prelims_Date varchar(15),
 Finals_Date varchar(15));

create table registrations
(First_Name varchar(20),
 Last_Name varchar(20),
 Grade integer,
 Event_Name varchar(25));

create table captains
(Admission_No integer Primary key,
 First_Name varchar(20),
 Last_Name varchar(20),
 House varchar(20));
